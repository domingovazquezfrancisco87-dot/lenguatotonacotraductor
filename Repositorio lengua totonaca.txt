const diccionarioBase=[

];
// QUITA DUPLICADOS - deja la última
const diccionario = [...new Map(diccionarioBase.map(o=>[o.tot.toLowerCase().trim(), o])).values()];

function normaliza(t){ return t.toLowerCase().replace(/_/g," ").replace(/\s+/g," ").trim(); }
let AUDIOS_NUEVOS = {};

async function localizarAudiosRecientes(){
 try{
  const res = await fetch(`https://api.github.com/repos/domingovazquezfrancisco87-dot/lenguatotonacotraductor/contents/audios?t=${Date.now()}`);
  const archivos = await res.json();
  const soloAudios = archivos.filter(a=>/\.(m4a|mp3|wav|ogg)$/i.test(a.name));
  soloAudios.forEach(a=>{ AUDIOS_NUEVOS[normaliza(a.name.replace(/\.[^/.]+$/, ""))] = `audios/${a.name}`; });
  document.getElementById("estado").innerText = `✅ ${soloAudios.length} audios recientes localizados automáticamente`;
  document.getElementById("contador").innerText = `${diccionario.length} palabras + ${soloAudios.length} audios reales`;
  renderListaReal();
 }catch(e){
  document.getElementById("estado").innerText = "Modo local";
  renderListaReal();
 }
}

function playAudio(tot){
 let url = AUDIOS_NUEVOS[normaliza(tot)];
 if(url){ new Audio(encodeURI(url)).play(); return; }
 let s=new SpeechSynthesisUtterance(tot); s.lang="es-MX"; s.rate=0.8; speechSynthesis.speak(s);
}

function traducirEspTot(){
 let q=document.getElementById("qEspTot").value.toLowerCase().trim();
 let div=document.getElementById("resEspTot");
 if(!q){div.innerHTML=`<span class="msg">Escribe en español</span>`; return;}
 let f=diccionario.filter(d=>d.esp.toLowerCase().includes(q));
 if(f.length==0){div.innerHTML=`<span class="msg">No encontrado: ${q}</span>`; return;}
 div.innerHTML=f.slice(0,15).map(d=>{
  let esReal=!!AUDIOS_NUEVOS[normaliza(d.tot)];
  return `<div class="row"><div class="tot"><button class="audio" onclick="playAudio('${d.tot.replace(/'/g,"\\'")}')">🔊</button> ${d.tot} <span class="badge ${esReal?'':'robot'}">${esReal?'VOZ REAL':'ROBOT'}</span></div><span class="esp">${d.esp}</span></div>`;
 }).join("");
}
function traducirTotEsp(){
 let q=document.getElementById("qTotEsp").value.toLowerCase().trim();
 let div=document.getElementById("resTotEsp");
 if(!q){div.innerHTML=`<span class="msg">Escribe en totonaco</span>`; return;}

 if(f.length==0){div.innerHTML=`<span class="msg">No encontrado: ${q}</span>`; return;}
 div.innerHTML=f.slice(0,15).map(d=>{
  let esReal=!!AUDIOS_NUEVOS[normaliza(d.tot)];
  return `<div class="row"><div class="tot"><button class="audio" onclick="playAudio('${d.tot.replace(/'/g,"\\'")}')">🔊</button> ${d.tot} <span class="badge ${esReal?'':'robot'}">${esReal?'VOZ REAL':'ROBOT'}</span></div><span class="esp">${d.esp}</span></div>`;
 }).join("");
}
function renderListaReal(){
 document.getElementById("listaReal").innerHTML = Object.keys(AUDIOS_NUEVOS).map(k=>{
  let d = diccionario.find(x=> normaliza(x.tot)===k );
  return `<div class="row"><div class="tot"><button class="audio" onclick="playAudio('${(d?d.tot:k).replace(/'/g,"\\'")}')">🔊</button> ${d?d.tot:k}</div><span class="esp">${d?d.esp:'audio reciente'}</span></div>`;
 }).join("") || `<div class="msg">Sube audios a /audios y aparecerán solos aquí</div>`;
}
localizarAudiosRecientes();
